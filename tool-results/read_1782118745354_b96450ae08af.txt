     1→'use client';
     2→
     3→import { useState, useRef, useCallback } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import { 
     6→  User, Lock, Building, MapPin, FileText, ArrowLeft, ArrowRight, 
     7→  AlertCircle, Loader2, Camera, Upload, X, Check, Eye, EyeOff,
     8→  Crop, RotateCcw, ZoomIn, ZoomOut
     9→} from 'lucide-react';
    10→import { Button } from '@/components/ui/button';
    11→import { Input } from '@/components/ui/input';
    12→import { Label } from '@/components/ui/label';
    13→import {
    14→  Card,
    15→  CardContent,
    16→  CardDescription,
    17→  CardHeader,
    18→  CardTitle,
    19→} from '@/components/ui/card';
    20→import { ThemeToggle } from '@/components/theme-toggle';
    21→import { useAuthStore } from '@/store/auth-store';
    22→import { CameraCapture } from '@/components/camera-capture';
    23→import { useLanguageStore } from '@/lib/i18n';
    24→
    25→interface AdminRegistrationProps {
    26→  onBack: () => void;
    27→  onRegistered: () => void;
    28→  initialPhone: string;
    29→}
    30→
    31→export function AdminRegistration({ onBack, onRegistered, initialPhone }: AdminRegistrationProps) {
    32→  const { t } = useLanguageStore();
    33→  const [step, setStep] = useState<1 | 2 | 3>(1);
    34→  const [isLoading, setIsLoading] = useState(false);
    35→  const [error, setError] = useState('');
    36→  const { login } = useAuthStore();
    37→  
    38→  // Step 1: Basic Details
    39→  const [formData, setFormData] = useState({
    40→    name: '',
    41→    phone: initialPhone,
    42→    email: '',
    43→    address: '',
    44→    organizationName: '',
    45→    organizationAddress: '',
    46→    gst: '',
    47→  });
    48→  
    49→  // Step 2: Profile Photo
    50→  const [profilePhoto, setProfilePhoto] = useState<string | null>(null);
    51→  const [showImageEditor, setShowImageEditor] = useState(false);
    52→  const [imageToEdit, setImageToEdit] = useState<string | null>(null);
    53→  const [cropArea, setCropArea] = useState({ x: 0, y: 0, size: 100 });
    54→  const [imageScale, setImageScale] = useState(1);
    55→  const [showCameraCapture, setShowCameraCapture] = useState(false);
    56→  const fileInputRef = useRef<HTMLInputElement>(null);
    57→  const canvasRef = useRef<HTMLCanvasElement>(null);
    58→  
    59→  // Step 3: User ID & Password
    60→  const [userId, setUserId] = useState('');
    61→  const [password, setPassword] = useState('');
    62→  const [confirmPassword, setConfirmPassword] = useState('');
    63→  const [securityPassword, setSecurityPassword] = useState('');
    64→  const [showPassword, setShowPassword] = useState(false);
    65→
    66→  const updateFormData = (field: string, value: string) => {
    67→    setFormData(prev => ({ ...prev, [field]: value }));
    68→    setError('');
    69→  };
    70→
    71→  const formatPhone = (value: string) => {
    72→    const numbers = value.replace(/\D/g, '');
    73→    if (numbers.length <= 3) return numbers;
    74→    if (numbers.length <= 6) return `${numbers.slice(0, 3)}-${numbers.slice(3)}`;
    75→    return `${numbers.slice(0, 3)}-${numbers.slice(3, 6)}-${numbers.slice(6, 10)}`;
    76→  };
    77→
    78→  // Step 1 Validation
    79→  const validateStep1 = () => {
    80→    if (!formData.name.trim()) {
    81→      setError(t.orgSetup.validationAdminName);
    82→      return false;
    83→    }
    84→    if (!formData.phone || formData.phone.length < 10) {
    85→      setError(t.orgSetup.validationAdminPhone);
    86→      return false;
    87→    }
    88→    if (!formData.organizationName.trim()) {
    89→      setError(t.orgSetup.validationOrgName);
    90→      return false;
    91→    }
    92→    if (!formData.organizationAddress.trim()) {
    93→      setError(t.orgSetup.validationOrgAddress);
    94→      return false;
    95→    }
    96→    return true;
    97→  };
    98→
    99→  // Step 2: Image handling
   100→  const handleImageSelect = (e: React.ChangeEvent<HTMLInputElement>) => {
   101→    const file = e.target.files?.[0];
   102→    if (file) {
   103→      const reader = new FileReader();
   104→      reader.onload = (event) => {
   105→        const dataUrl = event.target?.result as string;
   106→        setImageToEdit(dataUrl);
   107→        setShowImageEditor(true);
   108→        setCropArea({ x: 0, y: 0, size: 100 });
   109→        setImageScale(1);
   110→      };
   111→      reader.readAsDataURL(file);
   112→    }
   113→  };
   114→
   115→  const handleCapturePhoto = () => {
   116→    // Open the camera capture dialog
   117→    setShowCameraCapture(true);
   118→  };
   119→
   120→  const handleCameraCapture = (photo: string) => {
   121→    // Set the captured photo directly
   122→    setProfilePhoto(photo);
   123→    setShowCameraCapture(false);
   124→  };
   125→
   126→  const applyImageEdit = useCallback(() => {
   127→    if (!imageToEdit || !canvasRef.current) return;
   128→    
   129→    const canvas = canvasRef.current;
   130→    const ctx = canvas.getContext('2d');
   131→    if (!ctx) return;
   132→    
   133→    const img = new Image();
   134→    img.onload = () => {
   135→      const outputSize = 200;
   136→      canvas.width = outputSize;
   137→      canvas.height = outputSize;
   138→      
   139→      const minDim = Math.min(img.width, img.height);
   140→      const sx = cropArea.x * (img.width / 100);
   141→      const sy = cropArea.y * (img.height / 100);
   142→      const sSize = (cropArea.size / 100) * minDim;
   143→      
   144→      ctx.drawImage(img, sx, sy, sSize, sSize, 0, 0, outputSize, outputSize);
   145→      
   146→      const dataUrl = canvas.toDataURL('image/jpeg', 0.8);
   147→      setProfilePhoto(dataUrl);
   148→      setShowImageEditor(false);
   149→      setImageToEdit(null);
   150→    };
   151→    img.src = imageToEdit;
   152→  }, [imageToEdit, cropArea]);
   153→
   154→  const cancelImageEdit = () => {
   155→    setShowImageEditor(false);
   156→    setImageToEdit(null);
   157→  };
   158→
   159→  const removePhoto = () => {
   160→    setProfilePhoto(null);
   161→  };
   162→
   163→  // Step 3 Validation
   164→  const validateStep3 = () => {
   165→    if (!userId.trim()) {
   166→      setError(t.orgSetup.validationUserIdRequired);
   167→      return false;
   168→    }
   169→    if (userId.length < 4) {
   170→      setError(t.employee.validationUserId);
   171→      return false;
   172→    }
   173→    if (!password) {
   174→      setError(t.orgSetup.validationPasswordRequired);
   175→      return false;
   176→    }
   177→    if (password.length < 6) {
   178→      setError(t.employee.validationPassword);
   179→      return false;
   180→    }
   181→    if (password !== confirmPassword) {
   182→      setError(t.orgSetup.validationPasswordsMatch);
   183→      return false;
   184→    }
   185→    if (!securityPassword) {
   186→      setError(t.orgSetup.validationSecurityPasswordRequired);
   187→      return false;
   188→    }
   189→    if (securityPassword.length < 4) {
   190→      setError(t.employee.validationSecurityPassword);
   191→      return false;
   192→    }
   193→    return true;
   194→  };
   195→
   196→  // Navigation between steps
   197→  const goNext = () => {
   198→    setError('');
   199→    if (step === 1 && validateStep1()) {
   200→      setStep(2);
   201→    } else if (step === 2) {
   202→      setStep(3);
   203→      // Pre-fill userId from name
   204→      if (!userId) {
   205→        const baseUserId = formData.name.toLowerCase().replace(/\s+/g, '').slice(0, 8);
   206→        setUserId(baseUserId + Math.floor(Math.random() * 1000));
   207→      }
   208→    }
   209→  };
   210→
   211→  const goBack = () => {
   212→    setError('');
   213→    if (step === 1) {
   214→      onBack();
   215→    } else if (step === 2) {
   216→      setStep(1);
   217→    } else if (step === 3) {
   218→      setStep(2);
   219→    }
   220→  };
   221→
   222→  // Final Registration
   223→  const handleRegister = async () => {
   224→    if (!validateStep3()) return;
   225→    
   226→    setIsLoading(true);
   227→    setError('');
   228→
   229→    try {
   230→      const response = await fetch('/api/auth/register', {
   231→        method: 'POST',
   232→        headers: { 'Content-Type': 'application/json' },
   233→        body: JSON.stringify({
   234→          userId,
   235→          password,
   236→          securityPassword,
   237→          name: formData.name,
   238→          phone: formData.phone.replace(/\D/g, ''),
   239→          email: formData.email,
   240→          address: formData.address,
   241→          organizationName: formData.organizationName,
   242→          organizationAddress: formData.organizationAddress,
   243→          gst: formData.gst,
   244→          profilePhoto,
   245→        }),
   246→      });
   247→
   248→      const data = await response.json();
   249→
   250→      if (!response.ok) {
   251→        if (data.needsSetup) {
   252→          throw new Error('⚠️ Database not configured! Go to Vercel → Settings → Environment Variables → Add DATABASE_URL (PostgreSQL) and NEXTAUTH_SECRET, then redeploy.');
   253→        }
   254→        throw new Error(data.error || t.orgSetup.registrationFailed);
   255→      }
   256→
   257→      // Login the user
   258→      login({
   259→        id: data.user.id,
   260→        userId: data.user.userId,
   261→        phone: data.user.phone,
   262→        name: data.user.name,
   263→        role: 'admin',
   264→        organizationId: data.user.organizationId,
   265→        organizationName: data.user.organizationName,
   266→        email: data.user.email,
   267→        profilePhoto: data.user.profilePhoto,
   268→      });
   269→
   270→      setIsLoading(false);
   271→      onRegistered();
   272→
   273→    } catch (err: unknown) {
   274→      const errorMessage = err instanceof Error ? err.message : t.orgSetup.registrationFailedRetry;
   275→      setError(errorMessage);
   276→      setIsLoading(false);
   277→    }
   278→  };
   279→
   280→  return (
   281→    <div className="min-h-screen flex flex-col bg-gradient-to-br from-background via-background to-muted/20">
   282→      <div className="absolute top-4 right-4 z-10"><ThemeToggle /></div>
   283→
   284→      <div className="flex-1 flex flex-col items-center justify-center p-6">
   285→        <motion.div 
   286→          initial={{ opacity: 0, y: 20 }} 
   287→          animate={{ opacity: 1, y: 0 }} 
   288→          className="w-full max-w-md"
   289→        >
   290→          {/* Logo */}
   291→          <div className="flex flex-col items-center mb-6">
   292→            <motion.div 
   293→              initial={{ scale: 0 }} 
   294→              animate={{ scale: 1 }} 
   295→              className="w-14 h-14 rounded-2xl overflow-hidden shadow-xl shadow-emerald-500/20 mb-3 bg-white"
   296→            >
   297→              <img src="/logo.jpg" alt="Logo" className="w-full h-full object-cover" />
   298→            </motion.div>
   299→            <h1 className="text-xl font-bold">{t.orgSetup.adminRegistration}</h1>
   300→            <p className="text-muted-foreground text-sm mt-1">{t.orgSetup.createOrgAccount}</p>
   301→          </div>
   302→
   303→          {/* Progress Steps */}
   304→          <div className="flex items-center justify-center gap-2 mb-6">
   305→            {[1, 2, 3].map((s) => (
   306→              <div key={s} className="flex items-center">
   307→                <div className={`w-8 h-8 rounded-full flex items-center justify-center text-sm font-medium transition-all ${
   308→                  step >= s 
   309→                    ? 'bg-emerald-500 text-white' 
   310→                    : 'bg-muted text-muted-foreground'
   311→                }`}>
   312→                  {step > s ? <Check className="h-4 w-4" /> : s}
   313→                </div>
   314→                {s < 3 && (
   315→                  <div className={`w-8 h-0.5 mx-1 transition-all ${
   316→                    step > s ? 'bg-emerald-500' : 'bg-muted'
   317→                  }`} />
   318→                )}
   319→              </div>
   320→            ))}
   321→          </div>
   322→
   323→          {/* Step Labels */}
   324→          <div className="flex justify-between text-xs text-muted-foreground mb-4 px-2">
   325→            <span className={step >= 1 ? 'text-emerald-600 font-medium' : ''}>{t.orgSetup.stepDetails}</span>
   326→            <span className={step >= 2 ? 'text-emerald-600 font-medium' : ''}>{t.orgSetup.stepPhoto}</span>
   327→            <span className={step >= 3 ? 'text-emerald-600 font-medium' : ''}>{t.orgSetup.stepCredentials}</span>
   328→          </div>
   329→
   330→          <Card className="border-0 shadow-2xl bg-card/50 backdrop-blur-sm">
   331→            <CardHeader className="text-center pb-2">
   332→              <CardTitle className="text-lg">
   333→                {step === 1 && t.orgSetup.orgSetup}
   334→                {step === 2 && t.orgSetup.profilePhoto}
   335→                {step === 3 && t.orgSetup.setCredentials}
   336→              </CardTitle>
   337→              <CardDescription>
   338→                {step === 1 && t.orgSetup.fillOrgInfo}
   339→                {step === 2 && t.orgSetup.addProfilePhoto}
   340→                {step === 3 && t.orgSetup.createLoginCredentials}
   341→              </CardDescription>
   342→            </CardHeader>
   343→            <CardContent className="space-y-4">
   344→              {/* Step 1: Details */}
   345→              <AnimatePresence mode="wait">
   346→                {step === 1 && (
   347→                  <motion.div
   348→                    key="step1"
   349→                    initial={{ opacity: 0, x: -20 }}
   350→                    animate={{ opacity: 1, x: 0 }}
   351→                    exit={{ opacity: 0, x: 20 }}
   352→                    className="space-y-4"
   353→                  >
   354→                    <div className="space-y-2">
   355→                      <Label>{t.orgSetup.adminName} *</Label>
   356→                      <Input 
   357→                        placeholder={t.orgSetup.enterAdminName} 
   358→                        value={formData.name} 
   359→                        onChange={(e) => updateFormData('name', e.target.value)} 
   360→                        className="h-11" 
   361→                      />
   362→                    </div>
   363→                    <div className="space-y-2">
   364→                      <Label>{t.orgSetup.adminPhone} *</Label>
   365→                      <Input 
   366→                        placeholder={t.orgSetup.enterAdminPhone} 
   367→                        value={formatPhone(formData.phone)} 
   368→                        onChange={(e) => updateFormData('phone', e.target.value.replace(/\D/g, ''))} 
   369→                        className="h-11" 
   370→                        maxLength={12} 
   371→                      />
   372→                    </div>
   373→                    <div className="space-y-2">
   374→                      <Label>{t.orgSetup.adminEmail}</Label>
   375→                      <Input 
   376→                        type="email" 
   377→                        placeholder={t.orgSetup.enterAdminEmail} 
   378→                        value={formData.email} 
   379→                        onChange={(e) => updateFormData('email', e.target.value)} 
   380→                        className="h-11" 
   381→                      />
   382→                    </div>
   383→                    <div className="space-y-2">
   384→                      <Label>{t.orgSetup.orgName} *</Label>
   385→                      <div className="relative">
   386→                        <Building className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   387→                        <Input 
   388→                          placeholder={t.orgSetup.enterOrgName} 
   389→                          value={formData.organizationName} 
   390→                          onChange={(e) => updateFormData('organizationName', e.target.value)} 
   391→                          className="pl-10 h-11" 
   392→                        />
   393→                      </div>
   394→                    </div>
   395→                    <div className="space-y-2">
   396→                      <Label>{t.orgSetup.orgAddress} *</Label>
   397→                      <div className="relative">
   398→                        <MapPin className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   399→                        <Input 
   400→                          placeholder={t.orgSetup.enterOrgAddress} 
   401→                          value={formData.organizationAddress} 
   402→                          onChange={(e) => updateFormData('organizationAddress', e.target.value)} 
   403→                          className="pl-10 h-11" 
   404→                        />
   405→                      </div>
   406→                    </div>
   407→                    <div className="space-y-2">
   408→                      <Label>{t.orgSetup.orgGst}</Label>
   409→                      <div className="relative">
   410→                        <FileText className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   411→                        <Input 
   412→                          placeholder={t.orgSetup.enterGst} 
   413→                          value={formData.gst} 
   414→                          onChange={(e) => updateFormData('gst', e.target.value)} 
   415→                          className="pl-10 h-11" 
   416→                        />
   417→                      </div>
   418→                    </div>
   419→                  </motion.div>
   420→                )}
   421→
   422→                {/* Step 2: Profile Photo */}
   423→                {step === 2 && (
   424→                  <motion.div
   425→                    key="step2"
   426→                    initial={{ opacity: 0, x: -20 }}
   427→                    animate={{ opacity: 1, x: 0 }}
   428→                    exit={{ opacity: 0, x: 20 }}
   429→                    className="space-y-4"
   430→                  >
   431→                    {!showImageEditor ? (
   432→                      <>
   433→                        {/* Current Photo Display */}
   434→                        {profilePhoto ? (
   435→                          <div className="flex flex-col items-center gap-4">
   436→                            <div className="relative">
   437→                              <div className="w-32 h-32 rounded-full overflow-hidden border-4 border-emerald-500/30">
   438→                                <img src={profilePhoto} alt="Profile" className="w-full h-full object-cover" />
   439→                              </div>
   440→                              <button
   441→                                onClick={removePhoto}
   442→                                className="absolute -top-2 -right-2 w-8 h-8 bg-red-500 text-white rounded-full flex items-center justify-center hover:bg-red-600"
   443→                              >
   444→                                <X className="h-4 w-4" />
   445→                              </button>
   446→                            </div>
   447→                            <p className="text-sm text-muted-foreground">{t.orgSetup.profilePhotoSet}</p>
   448→                          </div>
   449→                        ) : (
   450→                          <div className="flex flex-col items-center gap-4 py-4">
   451→                            <div className="w-32 h-32 rounded-full bg-muted flex items-center justify-center border-2 border-dashed border-muted-foreground/30">
   452→                              <User className="h-12 w-12 text-muted-foreground/50" />
   453→                            </div>
   454→                            <p className="text-sm text-muted-foreground">{t.orgSetup.noProfilePhoto}</p>
   455→                          </div>
   456→                        )}
   457→
   458→                        {/* Upload/Capture Buttons */}
   459→                        <div className="grid grid-cols-2 gap-3">
   460→                          <Button
   461→                            variant="outline"
   462→                            onClick={() => fileInputRef.current?.click()}
   463→                            className="h-16 flex-col"
   464→                          >
   465→                            <Upload className="h-5 w-5 mb-1" />
   466→                            <span className="text-xs">{t.orgSetup.uploadPhoto}</span>
   467→                          </Button>
   468→                          <Button
   469→                            variant="outline"
   470→                            onClick={handleCapturePhoto}
   471→                            className="h-16 flex-col"
   472→                          >
   473→                            <Camera className="h-5 w-5 mb-1" />
   474→                            <span className="text-xs">{t.orgSetup.camera}</span>
   475→                          </Button>
   476→                        </div>
   477→                        <input
   478→                          ref={fileInputRef}
   479→                          type="file"
   480→                          accept="image/*"
   481→                          onChange={handleImageSelect}
   482→                          className="hidden"
   483→                        />
   484→                      </>
   485→                    ) : (
   486→                      /* Image Editor */
   487→                      <div className="space-y-4">
   488→                        <div className="relative w-full aspect-square bg-muted rounded-lg overflow-hidden">
   489→                          {imageToEdit && (
   490→                            <img 
   491→                              src={imageToEdit} 
   492→                              alt="Edit" 
   493→                              className="w-full h-full object-cover"
   494→                              style={{ transform: `scale(${imageScale})` }}
   495→                            />
   496→                          )}
   497→                          {/* Crop overlay */}
   498→                          <div className="absolute inset-0 flex items-center justify-center pointer-events-none">
   499→                            <div className="w-32 h-32 border-2 border-white rounded-full bg-black/20" />
   500→                          </div>
   501→                        </div>
   502→                        
   503→                        {/* Editor Controls */}
   504→                        <div className="flex justify-center gap-2">
   505→                          <Button
   506→                            variant="outline"
   507→                            size="icon"
   508→                            onClick={() => setImageScale(Math.max(0.5, imageScale - 0.1))}
   509→                          >
   510→                            <ZoomOut className="h-4 w-4" />
   511→                          </Button>
   512→                          <Button
   513→                            variant="outline"
   514→                            size="icon"
   515→                            onClick={() => setImageScale(Math.min(2, imageScale + 0.1))}
   516→                          >
   517→                            <ZoomIn className="h-4 w-4" />
   518→                          </Button>
   519→                          <Button
   520→                            variant="outline"
   521→                            size="icon"
   522→                            onClick={() => setImageScale(1)}
   523→                          >
   524→                            <RotateCcw className="h-4 w-4" />
   525→                          </Button>
   526→                        </div>
   527→                        
   528→                        <div className="flex gap-2">
   529→                          <Button variant="outline" onClick={cancelImageEdit} className="flex-1">
   530→                            {t.common.cancel}
   531→                          </Button>
   532→                          <Button onClick={applyImageEdit} className="flex-1 bg-emerald-500 hover:bg-emerald-600">
   533→                            <Check className="h-4 w-4 mr-2" />
   534→                            {t.orgSetup.apply}
   535→                          </Button>
   536→                        </div>
   537→                      </div>
   538→                    )}
   539→                    
   540→                    <canvas ref={canvasRef} className="hidden" />
   541→                  </motion.div>
   542→                )}
   543→
   544→                {/* Step 3: User ID & Password */}
   545→                {step === 3 && (
   546→                  <motion.div
   547→                    key="step3"
   548→                    initial={{ opacity: 0, x: -20 }}
   549→                    animate={{ opacity: 1, x: 0 }}
   550→                    exit={{ opacity: 0, x: 20 }}
   551→                    className="space-y-4"
   552→                  >
   553→                    <div className="space-y-2">
   554→                      <Label>{t.orgSetup.adminUserId} *</Label>
   555→                      <div className="relative">
   556→                        <User className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   557→                        <Input
   558→                          placeholder={t.orgSetup.createUserId}
   559→                          value={userId}
   560→                          onChange={(e) => {
   561→                            setUserId(e.target.value.toLowerCase().replace(/\s+/g, ''));
   562→                            setError('');
   563→                          }}
   564→                          className="pl-10 h-11"
   565→                        />
   566→                      </div>
   567→                      <p className="text-xs text-muted-foreground">{t.orgSetup.userIdMin4NoSpaces}</p>
   568→                    </div>
