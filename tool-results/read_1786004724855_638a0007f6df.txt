    65→  const [selectedEmployee, setSelectedEmployee] = useState<Employee | null>(null);
    66→  const [isSaving, setIsSaving] = useState(false);
    67→  const [error, setError] = useState('');

    73→    name: '',
    74→    phone: '',

    78→    department: '',
    79→    salary: '',
    80→    profilePhoto: '',

    83→    geofenceLat: '',

   103→

   113→  );
   114→

   119→      password: '',

   122→      email: '',
   123→      address: '',
   124→      designation: '',
   125→      department: '',
   126→      salary: '',
   127→      profilePhoto: '',
   128→      active: true,
   129→      geofenceEnabled: false,
   130→      geofenceLat: '',
   131→      geofenceLng: '',
   132→      geofenceRadius: '100',
   133→    });

   136→  };
   137→
   138→  // Open add dialog

   142→  };
   143→

   149→      password: '',

   151→      phone: employee.phone,
   152→      email: employee.email || '',
   153→      address: employee.address || '',
   154→      designation: employee.designation || '',
   155→      department: employee.department || '',
   156→      salary: employee.salary.toString(),
   157→      profilePhoto: employee.profilePhoto || '',
   158→      active: employee.active,
   159→      geofenceEnabled: (employee as any).geofenceEnabled || false,
   160→      geofenceLat: (employee as any).geofenceLat ? (employee as any).geofenceLat.toString() : '',

   179→        setFormData(prev => ({ ...prev, profilePhoto: event.target?.result as string }));
   180→      };

   198→    navigator.geolocation.getCurrentPosition(
   199→      (position) => {

   202→          geofenceLat: position.coords.latitude.toFixed(6),

   205→      },
   206→      (error) => {
   207→        setError('Unable to get your location. Please enter coordinates manually.');
   208→        console.error('Geolocation error:', error);
   209→      },
   210→      {
   211→        enableHighAccuracy: true,
   212→        timeout: 10000,
   213→        maximumAge: 0,
   214→      }

   216→  };

   222→      return;

   251→        headers: { 'Content-Type': 'application/json' },

   253→      });

   268→    }

   285→        id: selectedEmployee.id,

   287→        salary: parseFloat(formData.salary) || 0,
   288→      };
   289→
   290→      // Only include geofence fields if enabled

   297→        requestBody.geofenceEnabled = false;
   298→        requestBody.geofenceLat = null;
   299→        requestBody.geofenceLng = null;

   301→      }
   302→
   303→      const response = await fetch('/api/employees', {
   304→        method: 'PUT',
   305→        headers: { 'Content-Type': 'application/json' },
   306→        body: JSON.stringify(requestBody),
   307→      });
   308→
   309→      const data = await response.json();
   310→
   311→      if (!response.ok) {
   312→        throw new Error(data.error || 'Failed to update employee');
   313→      }

   315→      await fetchEmployees();
   316→      setShowEditDialog(false);
   317→      resetForm();
   318→    } catch (err: unknown) {
   319→      setError(err instanceof Error ? err.message : 'Failed to update employee');
   320→    } finally {
   321→      setIsSaving(false);
   322→    }
   323→  };

   337→        throw new Error('Failed to delete employee');

   611→        <div className="flex items-center gap-2 p-3 rounded-lg bg-red-500/10 border border-red-500/20">
   612→          <AlertCircle className="h-4 w-4 text-red-500" />
   613→          <p className="text-sm text-red-500">{error}</p>
   614→        </div>
   615→      )}
   616→    </div>
   617→  );
   618→
   619→  return (
   620→    <div className="space-y-4">
   621→      {/* Header */}
   622→      <div className="flex items-center justify-between">
   623→        <div>
   624→          <h2 className="text-xl font-bold">Employees</h2>
   625→          <p className="text-sm text-muted-foreground">{employees.length} employees registered</p>
   626→        </div>
   627→        <Button onClick={openAddDialog} className="bg-emerald-500 hover:bg-emerald-600">
   628→          <Plus className="h-4 w-4 mr-2" />
   629→          Add Employee
   630→        </Button>
   631→      </div>
   632→
   633→      {/* Search */}
   634→      <div className="relative">
   635→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   636→        <Input
   637→          value={searchQuery}
   638→          onChange={(e) => setSearchQuery(e.target.value)}
   639→          placeholder="Search by name, phone, or user ID..."
   640→          className="pl-9"
   641→        />
   642→      </div>
   643→
   644→      {/* Employee List */}
   645→      {isLoading ? (
   646→        <div className="flex items-center justify-center py-12">
   647→          <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   648→        </div>
   649→      ) : filteredEmployees.length === 0 ? (
   650→        <Card className="border-dashed">
   651→          <CardContent className="flex flex-col items-center justify-center py-12">
   652→            <User className="h-12 w-12 text-muted-foreground/50 mb-4" />
   653→            <p className="text-muted-foreground">No employees found</p>
   654→            <Button variant="link" onClick={openAddDialog}>Add your first employee</Button>
   655→          </CardContent>
   656→        </Card>
   657→      ) : (
   658→        <div className="grid gap-3">
   659→          {filteredEmployees.map((employee) => (
   660→            <motion.div
   661→              key={employee.id}
   662→              initial={{ opacity: 0, y: 10 }}
   663→              animate={{ opacity: 1, y: 0 }}
   664→              className="bg-card rounded-xl border p-4 flex items-center gap-4"
   665→            >
   666→              {/* Photo */}
   667→              {employee.profilePhoto ? (
   668→                <img 
   669→                  src={employee.profilePhoto} 
   670→                  alt={employee.name} 
   671→                  className="w-12 h-12 rounded-full object-cover"
   672→                />
   673→              ) : (
   674→                <div className="w-12 h-12 rounded-full bg-muted flex items-center justify-center">
   675→                  <User className="h-6 w-6 text-muted-foreground" />
   676→                </div>
   677→              )}
   678→
   679→              {/* Info */}
   680→              <div className="flex-1 min-w-0">
   681→                <div className="flex items-center gap-2">
   682→                  <p className="font-medium truncate">{employee.name}</p>
   683→                  {!employee.active && (
   684→                    <Badge variant="secondary" className="text-xs">Inactive</Badge>
   685→                  )}
   686→                </div>
   687→                <p className="text-sm text-muted-foreground">@{employee.userId}</p>
   688→                <div className="flex items-center gap-3 text-xs text-muted-foreground mt-1">
   689→                  <span>{employee.phone}</span>
   690→                  {employee.designation && <span>• {employee.designation}</span>}
   691→                </div>
   692→              </div>
   693→
   694→              {/* Salary */}
   695→              <div className="text-right hidden sm:block">
   696→                <p className="font-medium">₹{employee.salary.toLocaleString()}</p>
   697→                <p className="text-xs text-muted-foreground">/month</p>
   698→              </div>
   699→
   700→              {/* Actions */}
   701→              <div className="flex items-center gap-2">
   702→                <Button
   703→                  variant="ghost"
   704→                  size="icon"
   705→                  onClick={() => openEditDialog(employee)}
   706→                >
   707→                  <Edit2 className="h-4 w-4" />
   708→                </Button>
   709→                <Button
   710→                  variant="ghost"
   711→                  size="icon"
   712→                  onClick={() => openDeleteDialog(employee)}
   713→                  className="text-red-500 hover:text-red-600 hover:bg-red-500/10"
   714→                >
   715→                  <Trash2 className="h-4 w-4" />
   716→                </Button>
   717→              </div>
   718→            </motion.div>
   719→          ))}
   720→        </div>
   721→      )}
   722→
   723→      {/* Add Dialog */}
   724→      <Dialog open={showAddDialog} onOpenChange={setShowAddDialog}>
   725→        <DialogContent className="max-w-md">
   726→          <DialogHeader>
   727→            <DialogTitle>Add New Employee</DialogTitle>
   728→          </DialogHeader>
   729→          <FormFields />
   730→          <div className="flex gap-2 pt-4">
   731→            <Button variant="outline" onClick={() => setShowAddDialog(false)} className="flex-1">
   732→              Cancel
   733→            </Button>
   734→            <Button onClick={handleAddEmployee} disabled={isSaving} className="flex-1 bg-emerald-500 hover:bg-emerald-600">
   735→              {isSaving ? <Loader2 className="h-4 w-4 animate-spin" /> : 'Add Employee'}
   736→            </Button>
   737→          </div>
   738→        </DialogContent>
   739→      </Dialog>
   740→
   741→      {/* Edit Dialog */}
   742→      <Dialog open={showEditDialog} onOpenChange={setShowEditDialog}>
   743→        <DialogContent className="max-w-md">
   744→          <DialogHeader>
   745→            <DialogTitle>Edit Employee</DialogTitle>
   746→          </DialogHeader>
   747→          <FormFields isEdit />
   748→          <div className="flex gap-2 pt-4">
   749→            <Button variant="outline" onClick={() => setShowEditDialog(false)} className="flex-1">
   750→              Cancel
   751→            </Button>
   752→            <Button onClick={handleEditEmployee} disabled={isSaving} className="flex-1 bg-emerald-500 hover:bg-emerald-600">
   753→              {isSaving ? <Loader2 className="h-4 w-4 animate-spin" /> : 'Save Changes'}
   754→            </Button>
   755→          </div>
   756→        </DialogContent>
   757→      </Dialog>
   758→
   759→      {/* Delete Dialog */}
   760→      <AlertDialog open={showDeleteDialog} onOpenChange={setShowDeleteDialog}>
   761→        <AlertDialogContent>
   762→          <AlertDialogHeader>
   763→            <AlertDialogTitle>Delete Employee</AlertDialogTitle>
   764→            <AlertDialogDescription>
   765→              Are you sure you want to delete <strong>{selectedEmployee?.name}</strong>?
   766→              This action cannot be undone. All attendance and leave records will also be deleted.
   767→            </AlertDialogDescription>
   768→          </AlertDialogHeader>
   769→          <AlertDialogFooter>
   770→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   771→            <AlertDialogAction
   772→              onClick={handleDeleteEmployee}
   773→              className="bg-red-500 hover:bg-red-600"
   774→            >
   775→              {isSaving ? <Loader2 className="h-4 w-4 animate-spin" /> : 'Delete'}
   776→            </AlertDialogAction>
   777→          </AlertDialogFooter>
   778→        </AlertDialogContent>
   779→      </AlertDialog>
   780→
   781→      {/* Camera Capture Dialog */}
   782→      <CameraCapture
   783→        open={showCameraCapture}
   784→        onCapture={handleCameraCapture}
   785→        onClose={() => setShowCameraCapture(false)}
   786→        title="📸 Capture Profile Photo"
   787→      />
   788→    </div>
   789→  );
   790→}
