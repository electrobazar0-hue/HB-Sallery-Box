   144→function getSyncSourceLabel(source?: string | null): string | null {
   145→  if (!source) return null;
   146→  if (source === 'google-calendar') return 'Google';
   147→  if (source === 'nager-date') return 'Nager';
   148→  if (source === 'static-database') return 'Static';
   149→  return source;
   150→}
   151→
   152→function getSourceDisplayName(source: string): string {
   153→  if (source === 'google-calendar') return 'Google Calendar';
   154→  if (source === 'nager-date') return 'Nager.Date';
   155→  if (source === 'static-database') return 'Static Database';
   156→  return source;
