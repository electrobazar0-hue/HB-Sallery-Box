    48→

    50→  id: string;

    57→  isPaid: boolean;
    58→  isOptional: boolean;

    62→  status: string;

    64→

    66→  totalHolidays: number;
    67→  activeHolidays: number;
    68→  draftHolidays: number;
    69→  paidHolidays: number;
    70→  halfDayHolidays: number;
    71→  optionalHolidays: number;
    72→  compOffHolidays: number;
    73→}
    74→
    75→interface HolidayManagementProps {
    76→  organizationId: string;
    77→  adminId: string;

    79→

    82→  { value: 'festival', label: 'Festival', icon: PartyPopper, color: 'bg-purple-500' },
    83→  { value: 'weekly', label: 'Weekly Off', icon: Clock, color: 'bg-blue-500' },
    84→  { value: 'company', label: 'Company', icon: Building, color: 'bg-emerald-500' },
    85→  { value: 'emergency', label: 'Emergency', icon: AlertTriangle, color: 'bg-red-500' },
    86→];
    87→

    96→];
    97→
    98→const emptyFormData = {
    99→  holidayName: '',
   100→  date: '',
   101→  holidayType: 'company',
   102→  description: '',
   103→  allowPunch: false,
   104→  isHalfDay: false,
   105→  isPaid: true,
   106→  isOptional: false,
   107→  compensatoryOff: false,
   108→  isRecurring: false,
   109→  recurringDay: 0,
   110→  status: 'active',

   112→
   113→export function HolidayManagement({ organizationId, adminId }: HolidayManagementProps) {
   114→  const { t } = useLanguageStore();
   115→  const [holidays, setHolidays] = useState<Holiday[]>([]);
   116→  const [stats, setStats] = useState<HolidayStats | null>(null);
   117→  const [isLoading, setIsLoading] = useState(true);
   118→  const [isSyncing, setIsSyncing] = useState(false);
   119→  const [showAddDialog, setShowAddDialog] = useState(false);
   120→  const [editingHoliday, setEditingHoliday] = useState<Holiday | null>(null);
   121→  const [deletingHoliday, setDeletingHoliday] = useState<Holiday | null>(null);
   122→  const [error, setError] = useState<string | null>(null);
   123→  const [success, setSuccess] = useState<string | null>(null);
   124→  const [currentMonth, setCurrentMonth] = useState(new Date());
   125→
   126→  const [formData, setFormData] = useState({ ...emptyFormData });
   127→
   128→  // Fetch holidays
   129→  const fetchHolidays = async () => {
   130→    if (!organizationId) {
   131→      setError(t.holiday.organizationNotFound);
   132→      setIsLoading(false);
   133→      return;
   134→    }
   135→
   136→    try {
   137→      setError(null);
   138→      const response = await fetch(`/api/holidays?organizationId=${organizationId}`);
   139→      const data = await response.json();
   140→
   141→      if (data.success) {
   142→        setHolidays(data.holidays || []);
   143→        setStats(data.stats || null);
   144→      } else {
   145→        setError(data.error || t.holiday.failedFetchHolidays);
   146→      }
   147→    } catch (err) {
   148→      console.error('Error fetching holidays:', err);
   149→      setError(t.holiday.failedFetchConnection);
   150→    } finally {
   151→      setIsLoading(false);
   152→    }
   153→  };
   154→
   155→  useEffect(() => {
   156→    fetchHolidays();
   157→  }, [organizationId]);
   158→
   159→  // Sync with Google Calendar
   160→  const handleSyncGoogleCalendar = async () => {
   161→    if (!organizationId) {
   162→      setError(t.holiday.organizationNotFound);
   163→      return;
   164→    }
   165→
   166→    setIsSyncing(true);
   167→    setError(null);
   168→
   169→    try {
   170→      const response = await fetch('/api/holidays/sync', {
   171→        method: 'POST',
   172→        headers: { 'Content-Type': 'application/json' },
   173→        body: JSON.stringify({ organizationId, adminId, country: 'indian' }),
   174→      });
   175→
   176→      const data = await response.json();
   177→
   178→      if (data.success) {
   179→        setSuccess(`${t.holiday.syncIndianHolidays}: ${data.added || 0} (${data.skipped || 0} ${t.holiday.repeatEveryWeek})`);
   180→        fetchHolidays();
   181→      } else {
   182→        setError(data.error || 'Failed to sync holidays');
   183→      }
   184→    } catch (err) {
   185→      console.error('Error syncing holidays:', err);
   186→      setError(t.holiday.failedFetchConnection);
   187→    } finally {
   188→      setIsSyncing(false);
   189→    }
   190→  };
   191→
   192→  // Add/Edit holiday
   193→  const handleSaveHoliday = async () => {
   194→    if (!formData.holidayName || !formData.date) {
   195→      setError(t.holiday.holidayNameDateRequired);
   196→      return;
   197→    }
   198→
   199→    setIsLoading(true);
   200→    setError(null);
   201→
   202→    try {
   203→      const url = '/api/holidays';
   204→      const method = editingHoliday ? 'PUT' : 'POST';
   205→      const body = editingHoliday
   206→        ? { id: editingHoliday.id, ...formData }
   207→        : { organizationId, createdBy: adminId, ...formData };
   208→
   209→      const response = await fetch(url, {
   210→        method,
   211→        headers: { 'Content-Type': 'application/json' },
   212→        body: JSON.stringify(body),
   213→      });
   214→
   215→      const data = await response.json();
   216→
   217→      if (data.success) {
   218→        setSuccess(editingHoliday ? t.holiday.holidayUpdated : t.holiday.holidayAdded);
   219→        setShowAddDialog(false);
   220→        setEditingHoliday(null);
   221→        setFormData({ ...emptyFormData });
   222→        fetchHolidays();
   223→      } else {
   224→        setError(data.error || t.holiday.failedSaveHoliday);
   225→      }
   226→    } catch (err) {
   227→      console.error('Error saving holiday:', err);
   228→      setError(t.holiday.failedSaveTryAgain);
   229→    } finally {
   230→      setIsLoading(false);
   231→    }
   232→  };
   233→
   234→  // Quick toggle (allowPunch, isPaid, etc.) — directly update a single field
   235→  const handleQuickToggle = async (holiday: Holiday, field: string, value: boolean) => {
   236→    try {
   237→      const response = await fetch('/api/holidays', {
   238→        method: 'PUT',
   239→        headers: { 'Content-Type': 'application/json' },
   240→        body: JSON.stringify({
   241→          id: holiday.id,
   242→          holidayName: holiday.holidayName,
   243→          date: holiday.date,
   244→          holidayType: holiday.holidayType,
   245→          description: holiday.description || '',
   246→          allowPunch: field === 'allowPunch' ? value : holiday.allowPunch,
   247→          isHalfDay: field === 'isHalfDay' ? value : holiday.isHalfDay,
   248→          isPaid: field === 'isPaid' ? value : holiday.isPaid,
   249→          isOptional: field === 'isOptional' ? value : holiday.isOptional,
   250→          compensatoryOff: field === 'compensatoryOff' ? value : holiday.compensatoryOff,
   251→          isRecurring: holiday.isRecurring,
   252→          recurringDay: holiday.recurringDay || 0,
   253→          status: holiday.status,
   254→        }),
   255→      });
   256→
   257→      const data = await response.json();
   258→      if (data.success) {
   259→        fetchHolidays();
   260→        const labelMap: Record<string, string> = {
   261→          allowPunch: t.holiday.allowAttendance,
   262→          isHalfDay: t.holiday.halfDay,
   263→          isPaid: t.holiday.paid,
   264→          isOptional: t.holiday.optionalLabel,
   265→          compensatoryOff: t.holiday.compOffLabel,
   266→        };
   267→        setSuccess(`${holiday.holidayName}: ${labelMap[field] || field} ${value ? t.holiday.enabled : t.holiday.disabled}`);
   268→      }
   269→    } catch (err) {
   270→      setError(t.holiday.failedUpdateHoliday);
   271→    }
   272→  };
   273→
   274→  // Toggle publish/draft status
   275→  const handleTogglePublish = async (holiday: Holiday) => {
   276→    const newStatus = holiday.status === 'active' ? 'draft' : 'active';
   277→    try {
   278→      const response = await fetch('/api/holidays', {
   279→        method: 'PUT',
   280→        headers: { 'Content-Type': 'application/json' },
   281→        body: JSON.stringify({
   282→          id: holiday.id,
   283→          holidayName: holiday.holidayName,
   284→          date: holiday.date,
   285→          holidayType: holiday.holidayType,
   286→          description: holiday.description || '',
   287→          allowPunch: holiday.allowPunch,
   288→          isHalfDay: holiday.isHalfDay,
   289→          isPaid: holiday.isPaid,
   290→          isOptional: holiday.isOptional,
   291→          compensatoryOff: holiday.compensatoryOff,
   292→          isRecurring: holiday.isRecurring,
   293→          recurringDay: holiday.recurringDay || 0,
   294→          status: newStatus,
   295→        }),
   296→      });
   297→
   298→      const data = await response.json();
   299→      if (data.success) {
   300→        setSuccess(newStatus === 'active'
   301→          ? `${holiday.holidayName} - ${t.holiday.holidayPublished}`
   302→          : `${holiday.holidayName} - ${t.holiday.holidayHidden}`
   303→        );
   304→        fetchHolidays();
   305→      }
   306→    } catch (err) {
   307→      setError(t.holiday.failedUpdateHoliday);
   308→    }
   309→  };
   310→
   311→  // Delete holiday
   312→  const handleDeleteHoliday = async () => {
   313→    if (!deletingHoliday) return;
   314→
   315→    try {
   316→      const response = await fetch(`/api/holidays?id=${deletingHoliday.id}`, {
   317→        method: 'DELETE',
   318→      });
   319→
   320→      const data = await response.json();
   321→
   322→      if (data.success) {
   323→        setSuccess(t.holiday.holidayDeleted);
   324→        setDeletingHoliday(null);
   325→        fetchHolidays();
   326→      } else {
   327→        setError(data.error || t.holiday.failedDeleteHoliday);
   328→      }
   329→    } catch (err) {
   330→      console.error('Error deleting holiday:', err);
   331→      setError(t.holiday.failedDeleteTryAgain);
   332→    }
   333→  };
   334→
   335→  // Open edit dialog with holiday data
   336→  const openEditDialog = (holiday: Holiday) => {
   337→    setEditingHoliday(holiday);
   338→    setFormData({
   339→      holidayName: holiday.holidayName,
   340→      date: holiday.date,
   341→      holidayType: holiday.holidayType,
   342→      description: holiday.description || '',
   343→      allowPunch: holiday.allowPunch,
   344→      isHalfDay: holiday.isHalfDay,
   345→      isPaid: holiday.isPaid,
   346→      isOptional: holiday.isOptional,
   347→      compensatoryOff: holiday.compensatoryOff,
   348→      isRecurring: holiday.isRecurring,
   349→      recurringDay: holiday.recurringDay || 0,
   350→      status: holiday.status || 'active',
   351→    });
   352→    setShowAddDialog(true);
   353→  };
   354→
   355→  // Get holiday type info
   356→  const getHolidayTypeInfo = (type: string) => {
   357→    return holidayTypes.find(ht => ht.value === type) || holidayTypes[3];
   358→  };
   359→
   360→  // Calendar navigation
   361→  const prevMonth = () => setCurrentMonth(new Date(currentMonth.getFullYear(), currentMonth.getMonth() - 1, 1));
   362→  const nextMonth = () => setCurrentMonth(new Date(currentMonth.getFullYear(), currentMonth.getMonth() + 1, 1));
   363→
   364→  // Generate calendar days
   365→  const getCalendarDays = () => {
   366→    const year = currentMonth.getFullYear();
   367→    const month = currentMonth.getMonth();
   368→    const firstDay = new Date(year, month, 1);
   369→    const lastDay = new Date(year, month + 1, 0);
   370→    const daysInMonth = lastDay.getDate();
   371→    const startingDay = firstDay.getDay();
   372→    const today = new Date().toISOString().split('T')[0];
   373→
   374→    const days: Array<{ day: number; date: string; holiday?: Holiday | undefined; isToday: boolean } | null> = [];
   375→    for (let i = 0; i < startingDay; i++) days.push(null);
   376→    for (let i = 1; i <= daysInMonth; i++) {
   377→      const date = new Date(year, month, i);
   378→      const dateStr = date.toISOString().split('T')[0];
   379→      const holiday = holidays.find(h => h.date === dateStr);
   380→      const isToday = dateStr === today;
   381→      days.push({ day: i, date: dateStr, holiday, isToday });
   382→    }
   383→    return days;
   384→  };
   385→
   386→  // Get upcoming holidays
   387→  const getUpcomingHolidays = () => {
   388→    const today = new Date().toISOString().split('T')[0];
   389→    return holidays
   390→      .filter(h => h.date >= today && h.status === 'active')
   391→      .sort((a, b) => a.date.localeCompare(b.date))
   392→      .slice(0, 5);
   393→  };
   394→
   395→  // Clear messages after 5 seconds
   396→  useEffect(() => {
   397→    if (error || success) {
   398→      const timer = setTimeout(() => { setError(null); setSuccess(null); }, 5000);
   399→      return () => clearTimeout(timer);
   400→    }
   401→  }, [error, success]);
   402→
   403→  return (
   404→    <div className="space-y-4">
   405→      {/* Header Actions */}
   406→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   407→        <div className="flex flex-col sm:flex-row gap-2">
   408→          <Button
   409→            onClick={() => {
   410→              setEditingHoliday(null);
   411→              setFormData({ ...emptyFormData });
   412→              setShowAddDialog(true);
   413→            }}
   414→            className="bg-gradient-to-r from-emerald-500 to-teal-600 w-full sm:w-auto"
   415→          >
   416→            <Plus className="h-4 w-4 mr-2" />
   417→            {t.holiday.addHoliday}
   418→          </Button>
   419→          <Button
   420→            variant="outline"
   421→            onClick={handleSyncGoogleCalendar}
   422→            disabled={isSyncing}
   423→            className="w-full sm:w-auto"
   424→          >
   425→            {isSyncing ? (
   426→              <RefreshCw className="h-4 w-4 mr-2 animate-spin" />
   427→            ) : (
   428→              <Download className="h-4 w-4 mr-2" />
   429→            )}
   430→            {t.holiday.syncIndianHolidays}
   431→          </Button>
   432→        </div>
   433→        <div className="flex items-center gap-2">
   434→          <Badge variant="secondary" className="text-xs">
   435→            <Eye className="h-3 w-3 mr-1" />
   436→            {stats?.activeHolidays || 0} {t.holiday.employeesCanSee}
   437→          </Badge>
   438→          <Badge variant="outline" className="text-xs border-dashed">
   439→            <EyeOff className="h-3 w-3 mr-1" />
   440→            {stats?.draftHolidays || 0} {t.holiday.draft}
   441→          </Badge>
   442→        </div>
   443→      </div>
   444→
   445→      {/* Stats Cards */}
   446→      {stats && (
   447→        <div className="grid grid-cols-3 sm:grid-cols-4 lg:grid-cols-7 gap-2 sm:gap-3">
   448→          {[
   449→            { label: t.holiday.total, value: stats.totalHolidays, color: 'text-foreground' },
   450→            { label: t.holiday.paid, value: stats.paidHolidays, color: 'text-emerald-500' },
   451→            { label: t.holiday.halfDay, value: stats.halfDayHolidays, color: 'text-amber-500' },
   452→            { label: t.holiday.optionalLabel, value: stats.optionalHolidays, color: 'text-blue-500' },
   453→            { label: t.holiday.compOffLabel, value: stats.compOffHolidays, color: 'text-teal-500' },
   454→            { label: t.holiday.published, value: stats.activeHolidays, color: 'text-emerald-500' },
   455→            { label: t.holiday.draft, value: stats.draftHolidays, color: 'text-orange-400' },
   456→          ].map((stat) => (
   457→            <Card key={stat.label} className="border-0 shadow-sm py-2 px-3">
   458→              <p className={`text-lg sm:text-xl font-bold ${stat.color}`}>{stat.value}</p>
   459→              <p className="text-[10px] sm:text-xs text-muted-foreground">{stat.label}</p>
   460→            </Card>
   461→          ))}
   462→        </div>
   463→      )}
   464→
   465→      {/* Error/Success Messages */}
   466→      <AnimatePresence>
   467→        {error && (
   468→          <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }}>
   469→            <Alert variant="destructive">
   470→              <AlertTriangle className="h-4 w-4" />
   471→              <AlertDescription>{error}</AlertDescription>
   472→            </Alert>
   473→          </motion.div>
   474→        )}
   475→        {success && (
   476→          <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }}>
   477→            <Alert className="border-emerald-500 bg-emerald-500/10">
   478→              <Check className="h-4 w-4 text-emerald-500" />
   479→              <AlertDescription className="text-emerald-600 dark:text-emerald-400">{success}</AlertDescription>
   480→            </Alert>
   481→          </motion.div>
   482→        )}
   483→      </AnimatePresence>
   484→
   485→      {!organizationId && (
   486→        <Alert variant="destructive">
   487→          <AlertTriangle className="h-4 w-4" />
   488→          <AlertDescription>{t.holiday.organizationNotFound}</AlertDescription>
   489→        </Alert>
   490→      )}
   491→
   492→      {/* Main Grid */}
   493→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   494→        {/* Calendar View */}
   495→        <Card className="lg:col-span-2 border-0 shadow-md">
   496→          <CardHeader className="pb-2">
   497→            <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2">
   498→              <CardTitle className="text-base sm:text-lg flex items-center gap-2">
   499→                <CalendarDays className="h-4 w-4 sm:h-5 sm:w-5 text-emerald-500" />
   500→                {t.holiday.holidayCalendar}
   501→              </CardTitle>
   502→              <div className="flex items-center gap-1 sm:gap-2 self-center">
   503→                <Button variant="ghost" size="icon" onClick={prevMonth} className="h-8 w-8"><ChevronLeft className="h-4 w-4" /></Button>
   504→                <span className="font-medium text-sm sm:text-base min-w-[120px] sm:min-w-[150px] text-center">
   505→                  {currentMonth.toLocaleDateString('en-US', { month: 'long', year: 'numeric' })}
   506→                </span>
   507→                <Button variant="ghost" size="icon" onClick={nextMonth} className="h-8 w-8"><ChevronRight className="h-4 w-4" /></Button>
   508→              </div>
   509→            </div>
   510→          </CardHeader>
   511→          <CardContent>
   512→            <div className="grid grid-cols-7 gap-0.5 sm:gap-1 text-center mb-2">
   513→              {['S', 'M', 'T', 'W', 'T', 'F', 'S'].map((day, i) => (
   514→                <div key={i} className="text-[10px] sm:text-xs font-medium text-muted-foreground py-1 sm:py-2">{day}</div>
   515→              ))}
   516→            </div>
   517→            <div className="grid grid-cols-7 gap-0.5 sm:gap-1">
   518→              {getCalendarDays().map((day, index) => (
   519→                <div
   520→                  key={index}
   521→                  className={`aspect-square p-0.5 sm:p-1 rounded-lg text-xs sm:text-sm relative flex flex-col items-center justify-center cursor-pointer transition-all ${
   522→                    day?.isToday ? 'ring-2 ring-emerald-400' : ''
   523→                  } ${
   524→                    day?.holiday
   525→                      ? day.holiday.status === 'draft'
   526→                        ? 'bg-amber-100 dark:bg-amber-900/30 text-amber-600 border border-dashed border-amber-400'
   527→                        : day.holiday.isHalfDay
   528→                          ? 'bg-amber-400 text-white font-medium'
   529→                          : `${getHolidayTypeInfo(day.holiday.holidayType).color} text-white font-medium`
   530→                      : day ? 'hover:bg-muted' : ''
   531→                  }`}
   532→                  title={day?.holiday ? `${day.holiday.holidayName}${day.holiday.status === 'draft' ? ` (${t.holiday.draftHidden})` : ''}` : undefined}
   533→                  onClick={() => day?.holiday && openEditDialog(day.holiday)}
   534→                >
   535→                  {day?.day}
   536→                  {day?.holiday?.isHalfDay && day.holiday.status === 'active' && (
   537→                    <span className="text-[6px] sm:text-[8px] leading-none">1/2</span>
